IMPORT # ================================================================ import geopandas as gpd import pandas as pd import numpy as np import folium import html import os import json from google.colab import files from IPython.display import display, HTML from folium.plugins import Fullscreen, Search print("=" * 75) print(" BANDUNG DEMOGRAPHIC ATLAS 2024") print(" PROFESSIONAL GIS PORTFOLIO DASHBOARD") print("=" * 75) # ================================================================ # 3. VALIDATION # ================================================================ required = { "DESA": COL_DESA, "KECAMATAN": COL_KEC, "JUMLAH PENDUDUK 2024": COL_POP24, "KEPADATAN PENDUDUK 2024": COL_DENSITY } missing = [ key for key, value in required.items() if value is None ] if missing: raise Exception( "❌ Field penting tidak ditemukan:\n" + "\n".join(missing) ) # ================================================================ # 7. STANDARDIZE DATA # ================================================================ print("\n[3/12] Standardisasi data...") gdf["DESA_NAME"] = ( gdf[COL_DESA] .fillna("-") .astype(str) ) gdf["KEC_NAME"] = ( gdf[COL_KEC] .fillna("-") .astype(str) ) gdf["POP_2024"] = ( gdf[COL_POP24] .fillna(0) ) gdf["DENSITY_2024"] = ( gdf[COL_DENSITY] .fillna(0) ) # ------------------------------------------------ # Historical Population # ------------------------------------------------ for year, col in [ (2020, COL_POP20), (2021, COL_POP21), (2022, COL_POP22), (2023, COL_POP23) ]: if col is not None: gdf[ f"POP_{year}" ] = ( gdf[col] .fillna(0) ) else: gdf[ f"POP_{year}" ] = 0 # ------------------------------------------------ # Household # ------------------------------------------------ if COL_KK: gdf["KK"] = ( gdf[COL_KK] .fillna(0) ) else: gdf["KK"] = 0 # ------------------------------------------------ # Gender # ------------------------------------------------ if COL_MALE: gdf["MALE"] = ( gdf[COL_MALE] .fillna(0) ) else: gdf["MALE"] = 0 if COL_FEMALE: gdf["FEMALE"] = ( gdf[COL_FEMALE] .fillna(0) ) else: gdf["FEMALE"] = 0 # ------------------------------------------------ # Growth # ------------------------------------------------ if COL_GROWTH: gdf["GROWTH"] = ( gdf[COL_GROWTH] .fillna(0) ) else: gdf["GROWTH"] = np.where( gdf["POP_2020"] > 0, ( ( gdf["POP_2024"] - gdf["POP_2020"] ) / gdf["POP_2020"] ) * 100, 0 ) # ================================================================ # 8. CRS # ================================================================ print("\n[4/12] Menyiapkan CRS...") if gdf.crs is None: gdf = gdf.set_crs( epsg=4326 ) gdf = gdf.to_crs( epsg=4326 ) print("✓ CRS:", gdf.crs) # ================================================================ # 9. STATISTICS # ================================================================ print("\n[5/12] Menghitung statistik...") TOTAL_POP = ( gdf["POP_2024"] .sum() ) TOTAL_KK = ( gdf["KK"] .sum() ) AVG_DENSITY = ( gdf["DENSITY_2024"] .replace( [np.inf, -np.inf], np.nan ) .mean() ) AVG_GROWTH = ( gdf["GROWTH"] .replace( [np.inf, -np.inf], np.nan ) .mean() ) TOTAL_MALE = ( gdf["MALE"] .sum() ) TOTAL_FEMALE = ( gdf["FEMALE"] .sum() ) if TOTAL_POP > 0: MALE_PERCENT = ( TOTAL_MALE / TOTAL_POP * 100 ) FEMALE_PERCENT = ( TOTAL_FEMALE / TOTAL_POP * 100 ) else: MALE_PERCENT = 0 FEMALE_PERCENT = 0 # ================================================================ # 10. TOP 10 # ================================================================ print("\n[6/12] Membuat ranking...") top10 = ( gdf[ [ "DESA_NAME", "KEC_NAME", "POP_2024", "DENSITY_2024" ] ] .sort_values( "DENSITY_2024", ascending=False ) .head(10) .reset_index(drop=True) ) top10_html = "" for i, row in top10.iterrows(): desa = html.escape( str(row["DESA_NAME"]) ) kec = html.escape( str(row["KEC_NAME"]) ) top10_html += f"""

""" # ================================================================ # 11. CREATE FOLIUM MAP # ================================================================ m = folium.Map( location=[ MAP_LAT, MAP_LON ], zoom_start=11, tiles=None, control_scale=True ) # ================================================================ # BASEMAP 1 — LIGHT # ================================================================ folium.TileLayer( tiles="CartoDB positron", name="Light Map", control=True, overlay=False ).add_to(m) # ================================================================ # BASEMAP 2 — OSM # ================================================================ folium.TileLayer( tiles="OpenStreetMap", name="OpenStreetMap", control=True, overlay=False ).add_to(m) # ================================================================ # BASEMAP 3 — GOOGLE SATELLITE # ================================================================ folium.TileLayer( tiles= "https://mt1.google.com/vt/lyrs=s&x={x}&y={y}&z={z}", attr= "Google Satellite", name="Google Satellite", control=True, overlay=False, max_zoom=20 ).add_to(m) # ================================================================ # DENSITY RANGE # ================================================================ values = ( gdf["DENSITY_2024"] .replace( [np.inf, -np.inf], np.nan ) .fillna(0) ) vmin = float( values.min() ) vmax = float( values.max() ) if vmax <= vmin: vmax = vmin + 1 colors = [ "#f7fbff", "#deebf7", "#c6dbef", "#9ecae1", "#6baed6", "#4292c6", "#2171b5", "#08519c", "#08306b" ] def get_color(value): try: value = float( value ) except: value = 0 ratio = ( value - vmin ) / ( vmax - vmin ) ratio = max( 0, min(1, ratio) ) index = int( ratio * (len(colors)-1) ) return colors[index] # ================================================================ # STYLE # ================================================================ def style_function(feature): value = ( feature["properties"] .get( "DENSITY_2024", 0 ) ) return { "fillColor": get_color(value), "color": "#374151", "weight": 0.7, "fillOpacity": 0.78 } def highlight_function(feature): return { "weight": 3, "color": "#111827", "fillOpacity": 0.92 } # ================================================================ # GEOJSON LAYER # ================================================================ geojson = folium.GeoJson( gdf.to_json(), name= "Population Density 2024", style_function= style_function, highlight_function= highlight_function ) # ================================================================ # POPUP # ================================================================ for feature in geojson.data["features"]: props = feature["properties"] desa = html.escape( str( props.get( "DESA_NAME", "-" ) ) ) kec = html.escape( str( props.get( "KEC_NAME", "-" ) ) ) pop = float( props.get( "POP_2024", 0 ) or 0 ) density = float( props.get( "DENSITY_2024", 0 ) or 0 ) kk = float( props.get( "KK", 0 ) or 0 ) growth = float( props.get( "GROWTH", 0 ) or 0 ) male = float( props.get( "MALE", 0 ) or 0 ) female = float( props.get( "FEMALE", 0 ) or 0 ) popup_html = f"""

""" popup = folium.Popup( popup_html, max_width=350 ) # create geometry layer popup # popup is attached after layer creation below feature["properties"]["POPUP_HTML"] = popup_html # ================================================================ # ADD GEOJSON WITH POPUP + TOOLTIP # ================================================================ geojson = folium.GeoJson( gdf.to_json(), name= "Population Density 2024", style_function= style_function, highlight_function= highlight_function, tooltip=folium.GeoJsonTooltip( fields=[ "DESA_NAME", "KEC_NAME", "POP_2024", "DENSITY_2024" ], aliases=[ "Desa/Kelurahan", "Kecamatan", "Penduduk 2024", "Kepadatan" ], localize=True, sticky=False, labels=True, style=""" background-color:white; color:#111827; font-family:Arial; font-size:12px; padding:8px; border-radius:7px; """ ) ) # ================================================================ # ATTACH POPUP # ================================================================ for feature in geojson.data["features"]: popup_html = feature[ "properties" ].get( "POPUP_HTML", "" ) folium.Popup( popup_html, max_width=350 ).add_to( folium.GeoJson( { "type": "FeatureCollection", "features": [feature] }, style_function= style_function, highlight_function= highlight_function ) ) # ------------------------------------------------ # IMPORTANT: # Use a single GeoJson layer with popup binding # ------------------------------------------------ # Rebuild clean layer using on_each_feature def on_each_feature( feature, layer ): popup_html = ( feature["properties"] .get( "POPUP_HTML", "" ) ) layer.bindPopup( popup_html, maxWidth=350 ) geojson = folium.GeoJson( gdf.to_json(), name= "Population Density 2024", style_function= style_function, highlight_function= highlight_function, tooltip=folium.GeoJsonTooltip( fields=[ "DESA_NAME", "KEC_NAME", "POP_2024", "DENSITY_2024" ], aliases=[ "Desa/Kelurahan", "Kecamatan", "Penduduk 2024", "Kepadatan" ], localize=True, sticky=False, labels=True ), popup=folium.GeoJsonPopup( fields=[ "DESA_NAME", "KEC_NAME", "POP_2024", "DENSITY_2024", "KK", "GROWTH", "MALE", "FEMALE" ], aliases=[ "Desa/Kelurahan", "Kecamatan", "Penduduk 2024", "Kepadatan", "Jumlah KK", "Pertumbuhan (%)", "Laki-laki", "Perempuan" ], localize=True, labels=True, sticky=False ) ) geojson.add_to(m) # ================================================================ # SEARCH # ================================================================ print("\n[8/12] Menambahkan search lokasi...") Search( layer=geojson, geom_type="Polygon", placeholder= "Cari Desa/Kelurahan...", collapsed=False, search_label= "DESA_NAME", position= "topright" ).add_to(m) # ================================================================ # FULLSCREEN # ================================================================ Fullscreen( position="topright", title="Full Screen", title_cancel="Exit Full Screen", force_separate_button=True ).add_to(m) # ================================================================ # LEGEND # ================================================================ legend_html = f"""

""" m.get_root().html.add_child( folium.Element( legend_html ) ) # ================================================================ # LAYER CONTROL # ================================================================ folium.LayerControl( position="topright", collapsed=False ).add_to(m) # ================================================================ # FIT MAP # ================================================================ bounds = gdf.total_bounds m.fit_bounds( [ [ bounds[1], bounds[0] ], [ bounds[3], bounds[2] ] ] ) # ================================================================ # 13. POPULATION TREND # ================================================================ print("\n[9/12] Membuat statistik...") years = [ 2020, 2021, 2022, 2023, 2024 ] population_values = [ int( gdf[ f"POP_{year}" ].sum() ) for year in years ] max_population = max( population_values ) trend_html = "" for year, value in zip( years, population_values ): if max_population > 0: width = ( value / max_population * 100 ) else: width = 0 trend_html += f"""

""" # ================================================================ # 14.
